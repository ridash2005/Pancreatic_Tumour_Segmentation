# 2. ENVIRONMENT SETUP

import os

# Set base directory (Update this to your project path in Colab)
# If you uploaded the folder, it might be '/content/Pancreatic_Tumour_Segmentation'
BASE_DIR = os.getcwd() 

os.environ['nnUNet_raw'] = os.path.join(BASE_DIR, 'nnUNet_raw')
os.environ['nnUNet_preprocessed'] = os.path.join(BASE_DIR, 'nnUNet_preprocessed')
os.environ['nnUNet_results'] = os.path.join(BASE_DIR, 'nnUNet_results')

# Create necessary directories
os.makedirs(os.environ['nnUNet_preprocessed'], exist_ok=True)
os.makedirs(os.environ['nnUNet_results'], exist_ok=True)

print("Environment Variables Set:")
print(f"nnUNet_raw: {os.environ['nnUNet_raw']}")
print(f"nnUNet_preprocessed: {os.environ['nnUNet_preprocessed']}")
print(f"nnUNet_results: {os.environ['nnUNet_results']}")


# ---------------------------------------------------------
# 3. PLANNING AND PREPROCESSING
# ---------------------------------------------------------
# For Task 1 (Dataset 501)
!nnUNetv2_plan_and_preprocess -d 501 -c 3d_fullres --verify_dataset_integrity


# ---------------------------------------------------------
# 4. TRAINING (Fold 0)
# ---------------------------------------------------------
# This command starts training for the first fold. 
# It takes a long time. Use --npz to save softmax for later.

# ---------------------------------------------------------
# 5. INFERENCE (PREDICTION)
# ---------------------------------------------------------
# Predict on the unlabeled test images (imagesTs)
# Ensure you have trained at least Fold 0.

os.makedirs(OUTPUT_FOLDER, exist_ok=True)

!nnUNetv2_predict -i {INPUT_FOLDER} -o {OUTPUT_FOLDER} -d 501 -c 3d_fullres -f 0 --save_probabilities
