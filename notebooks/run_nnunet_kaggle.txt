WORKING_DIR = '/kaggle/working'

def find_dataset_dir(search_path='/kaggle/input', dataset_id=501):
    """
    Finds the directory containing 'DatasetXXX' folders.
    """
    for root, dirs, files in os.walk(search_path):
        for d in dirs:
            if d.startswith(f'Dataset{dataset_id}'):
                return root
    return None

source_raw_root = find_dataset_dir()

if source_raw_root:
    print(f"Found nnUNet source raw root at: {source_raw_root}")
    
    raw_mirror_root = os.path.join(WORKING_DIR, 'nnUNet_raw')
    os.makedirs(raw_mirror_root, exist_ok=True)
    
    for item in os.listdir(source_raw_root):
        if item.startswith('Dataset'):
            s_dataset = os.path.join(source_raw_root, item)
            d_dataset = os.path.join(raw_mirror_root, item)
            
            os.makedirs(d_dataset, exist_ok=True)
            print(f"Mirroring {item}...")
            
            # Detect actual extension from imagesTr
            s_images_tr = os.path.join(s_dataset, 'imagesTr')
            detected_extension = ".nii.gz" # Default
            if os.path.exists(s_images_tr):
                files = [f for f in os.listdir(s_images_tr) if f.endswith('.nii') or f.endswith('.nii.gz')]
                if files:
                    if files[0].endswith('.nii.gz'):
                        detected_extension = ".nii.gz"
                    elif files[0].endswith('.nii'):
                        detected_extension = ".nii"
                    print(f"  Detected file extension: {detected_extension}")

            for sub in os.listdir(s_dataset):
                s_sub = os.path.join(s_dataset, sub)
                d_sub = os.path.join(d_dataset, sub)
                
                if sub == 'dataset.json':
                    # Fix: Dataset501 (PantherTask1) actually has labels 1 AND 2.
                    # Based on error logs: Found [0. 1. 2.]
                    # Label 1: Pancreas, Label 2: Tumour
                    print(f"  Updating dataset.json with correct extension and labels...")
                    with open(s_sub, 'r') as f:
                        data = json.load(f)
                    
                    data['file_ending'] = detected_extension
                    
                    # Fix Labels for Panther Task 1
                    if "501" in item:
                        data['labels'] = {
                            "background": 0,
                            "pancreas": 1,
                            "tumour": 2
                        }
                    # Fix Labels for Panther Task 2 (assuming identical structure)
                    elif "502" in item:
                         data['labels'] = {
                            "background": 0,
                            "pancreas": 1,
                            "tumour": 2
                        }
                        
                    with open(d_sub, 'w') as f:
                        json.dump(data, f, indent=4)
                elif os.path.isdir(s_sub):
                    # Folder: Create a symlink
                    if not os.path.exists(d_sub):
                        os.symlink(s_sub, d_sub)
                else:
                    # Other files: Copy
                    if not os.path.exists(d_sub):
                        shutil.copy2(s_sub, d_sub)

    os.environ['nnUNet_raw'] = raw_mirror_root
    print(f"Set nnUNet_raw to: {os.environ['nnUNet_raw']}")

else:
    print("CRITICAL: No Dataset folder found in /kaggle/input!")
    os.environ['nnUNet_raw'] = os.path.join(WORKING_DIR, 'nnUNet_raw')

os.environ['nnUNet_preprocessed'] = os.path.join(WORKING_DIR, 'nnUNet_preprocessed')
os.environ['nnUNet_results'] = os.path.join(WORKING_DIR, 'nnUNet_results')

os.makedirs(os.environ['nnUNet_preprocessed'], exist_ok=True)
